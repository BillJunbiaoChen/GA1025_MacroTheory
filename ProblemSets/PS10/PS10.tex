\section{Exercise RMT5 10.1 Equilibrium with valued fiat money}
\subsection{(a)}
Under CRRA utility and $beta = 1$, 
the optimal consumptions satisfy 
\[
{c^i_i}^{-\gamma} = {c^i_{i+1}}^{-\gamma}
\]
Therefore, using budget constraint, we have 
\[
c^i_i = c^i_{i+1} = \frac{1}{2}(\omega_1 + \frac{p_{i+1} \omega_2}{p_i})
\]
It follows that the saving function for a young generation is given by 
\begin{align*}
s(\omega_1, \omega_2, p_i, p_{i+1}) & = \omega_1 - \frac{1}{2}(\omega_1 + \frac{p_{i+1} \omega_2}{p_i}) \\ 
    & = \frac{1}{2}(\omega_1 - \frac{p_{i+1} \omega_2}{p_i}) 
\end{align*}


\subsection{(b)}
\textbf{Definition}
An equilibrium with valued fiat money is given by an allocation and a price sequence $p_t$,
such that, 
given the price sequence, the allocation solves the utility maximization problem for each agent $i \geq 1$.
\begin{align*}
    \max_{c_i^i, c_{i+1}^i} \quad \frac{{c^i_i}^{1-\gamma}}{1 - \gamma} & + \frac{{c^i_{i+1}}^{1-\gamma}}{1 - \gamma} \\ 
    \text{subject to: } \quad c_i^i + \frac{m_i}{p_i} & = \omega_1 \\ 
    c_i^{i+1}  & = \omega_2 + \frac{m_i}{p_{i+1}} \\ 
    m_i & \geq 0
\end{align*}
