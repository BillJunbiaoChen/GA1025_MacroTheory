\subsection{Exercise 7.1}
\textbf{A McCall Model with a chance of unemployment.}
The Bellman equation is as follows
\begin{equation}
    v(w) & = \max_{a \in \{ \text{Accept, Reject} \}} \{ \frac{w}{1 - \beta}, \beta (\phi (c + \bar{V}) + (1 - \phi) \mathbb{E} \left[ v(w') \right]  ) \}
\end{equation}
where $\bar{V} = \frac{c + \beta (1 - \phi) \mathbb{E} \left[ v(w') \right]}{1 - \beta \phi }$ 
denotes the continuation value of being unemployed.%
\footnote{
  I obtain $\bar{V}$ by solving the following Bellman equation:
  $\bar{V}  = c + \beta \bigg( \phi \bar{V} + (1 - \phi) \mathbb{E} \left[ v(w')\right]  \bigg)$
}
$v(w)$ denotes the value function conditional on receiving an offer.
$\mathbb{E}\left[ v(w') \right] = \int_{0}^B v(w') dF(w')$ denotes the expected value conditional on receiving an offer.


\subsection{Exercise 7.2}
\textbf{A McCall Model with two offers per period.}
Since $F(x) = \text{Pr}(z \leq x)$,  
the cumulative distribution function of the maximum of two random variables i.i.d. drawn from $F(x)$ is  
$G(x_m) = \text{Pr}(x < x_m) \cdot \text{Pr}(x < x_m) = [F(x_m)]^2$.  

\begin{equation}
    v(w_m) = \max_{a \in \{\text{Accept, Reject} \}}  
    \bigg( \frac{w_m}{1 - \beta}, c + \beta \mathbb{E}_{\max}[v(w_m')] \bigg)
\end{equation}
where $\mathbb{E}_{\max}[v(w_m')] = \int_0^{B} v(w_m') \, d[F(w_m')]^2$.  
Since $\mathbb{E}_{\max}[v(w_m')] = 2 \int_0^{B} v(w_m') f(w_m') F(w_m') dw' > \int_0^{B} v(w_m') f(w_m') dw'$,%
\footnote{
We apply the monotonicity of $v(x)$ and $2F(x)-1$ to finish the proof:
\begin{align*}
  & 2 \int_0^{B} v(w_m') f(w_m') F(w_m') dw' - \int_0^{B} v(w_m') f(w_m') dw' \\
  = & \int_0^{B} v(w_m') (2F(w_m') - 1) dF(w') \\
  = & \mathbb{E}\left[ v(w_m') (2F(w_m') - 1) \right] \\
  = & \underbrace{\text{Cov}\left[ v(w_m'), 2F(w_m') - 1 \right]}_{> 0} + \underbrace{\mathbb{E}[v(w_m')] \mathbb{E}[2F(w_m') - 1]}_{>0} > 0
\end{align*}


}  
we have  
\[
\bar{w}_{\max} = (1 - \beta) \big(c + \beta \mathbb{E}_{\max}[v(w_m')]\big)  
> (1 - \beta) \big(c + \beta \mathbb{E}[v(w_m')]\big) = \bar{w}.
\]  
\(\blacksquare\)

\subsection{Exercise 7.4}
\textbf{A McCall Model with a random number of offers per period.}
Denote the probability of $n$ in this period but $m$ in the next period as $\pi_{mn}$.

\paragraph{Part a.} The Bellman Equation is given as follows.
\begin{equation}
  v(w,n) = \max_{\text{\{Accept, Reject\}}} \bigg(\frac{w}{1 - \beta}, c + \beta(\mathbb{E}[v(w',n')]) \bigg)
\end{equation}
where $w$ is the best offer.
$\mathbb{E}[v(w',n')] = \sum_{m=1}^N \pi_{mn} \int_{0}^{B} m F(w')^{m-1}v(w', n' = m) dw'$ with $\sum_m \pi_{mn} = 1$.

\paragraph{Part b.} We characterize the reservation wage $\bar{w}(n)$ as follows.
\begin{equation}
  \bar{w}(n) = (1 - \beta)\beta \bigg[ \sum_{m=1}^N \pi_{mn} \int_{0}^{B} v(w', n'=m) dF(w')\bigg] + c(1-\beta)
\end{equation}
where $\sum_{m=1}^N \pi_{mn} \int_{0}^{B} v(w', n'=m) dF(w') \equiv \mathbb{E}[v(w', m)]$ is the marginal expected value function given the number of offers arrived.

Note that $\sum_m \pi_{mn} \mathbb{E}[v(w', m)]$ is a function of $n$, it follows that $\bar{w}(n)$ is also a function of $n$.

\subsection{Exercise 7.7}
\textbf{A McCall Model with variable labor supply.}
We first show that the optimal labor supply is a time invariant policy function, which solely depends on $w$.
Let $\tilde{v}(w)$ be the value function of accepting an offer at wage $w$.
Since the state variable $w$ does not change across periods, the Bellman equation that characterize the optimal labor supply is given by:
\begin{equation}
  \tilde{v}(w) = \max_{l \in [0, 1]} [ u(w(1-l), l) + \beta \tilde{v}(w)]
\end{equation}
Denote the optimal labor supply given wage $w$ by $l^*(w)$, we have 
\[
u(w(1-l^*), l^*) = (1 - \beta) \tilde{v}^*(w)
\]
Therefore, the optimal labor supply $l^*(w)$ does not depend on time (because $\tilde{v}^*(w)$ is time-invariant.)%
\footnote{
  Here I assume that the worker will stick to one optimal labor supply level under multiple equilibria. 
}

Finally, we can specify the Bellman equation using the optimal labor supply policy $l^*(w)$ as follows.
\begin{equation}
  v(w) = \max_{\text{\{Accept, Reject\}}} \Bigg\{\frac{u \bigg(w(1-l^*(w)), l^*(w)\bigg)}{1 - \beta}, u(c, 1) + \beta \int_{0}^B v(w') dF(w') \bigg\}
\end{equation}
Note that the preservation wage structure in the Bellman equation above is given by following equation.
\begin{equation}
  u \bigg(w(1-l^*(w)), l^*(w)\bigg) = (1-\beta)(u(c,1) + \beta \mathbb{E}[v(w')])
\end{equation}

\end{equation}

\section{Mean-preserving Spread}
\subsection{Answer to Question 2.1}
\paragraph{(a)}
According to the definition of mean-perserving spread (MSP) in \cite{Rothschild:1978},
$Y$ is a MSP from $X$ if $Y = X + Z$ where $\mathbb{E}[Z|X] = 0$.
Therefore, 
\begin{align*}
  \text{Var}(Y) & = \text{Var}(X + Z) \\ 
  & = \text{Var}(X + Z) \\
  & = \text{Var}(X) + \text{Var}(Z) + 2 \text{Cov}(X, Z) \\
  & = \text{Var}(X) + \text{Var}(Z) + 2 \bigg(\mathbb{E}(XZ) - \mathbb{E}(X) \mathbb{E}(Z)\bigg) \\
  & = \text{Var}(X) + \text{Var}(Z) + 2 \bigg(\mathbb{E}(X\mathbb{E}(Z|X)) - \mathbb{E}(X) \mathbb{E}(\mathbb{E}[Z|X])\bigg) \\
  & = \text{Var}(X) + \text{Var}(Z) \\
  & \geq \text{Var}(X)
\end{align*}
\(\blacksquare\)

\paragraph{(b)}
$Y$ can be generated by $X$ without using MSP while satisfies the properties of $\mathbb{E}(X) =\mathbb{E}(Y)$ and 
$\text{Var}(X) = \text{Var}(Y)$ as follows.
\begin{equation}
  \begin{bmatrix}
    X \\ Y
  \end{bmatrix}
  \sim N\left(
  \begin{bmatrix}
    1 \\ 0
  \end{bmatrix},
  \begin{bmatrix}
    1 & 0.5 \\ 
    0.5 & 1
  \end{bmatrix}
  \right)
\end{equation}
\textbf{Remark}: The process above is not a MSP because $\mathbb{E}[Z|X] \neq 0$.

\subsection*{Transitivity of mean-preserving spreads.}
\paragraph{(a)} Suppose $X2$ is contructed from $X1$ by MSP, and $X3$ is contructed from $X2$ by MSP.
Then we have
\begin{align*}
X2 &= X1 + Z1 \\
X3 &= X2 + Z2
\end{align*}
where $\mathbb{E}[Z1|X1] = 0$, and $\mathbb{E}[Z2|X2] = 0$.
Therefore, 
\[
X3 = X1 + Z1 + Z2
\]
Now if follows that 
\begin{align*}
\mathbb{E}[(Z1 + Z2)|X1] & = \mathbb{E}[(Z1 + Z2)|X1]  \\
& = \mathbb{E}[Z2|X1] \\
& = \mathbb{E}[Z2|X2, Z1] \\
& = \mathbb{E}[Z2|X2] (\text{because } Z1 \text{ is independent of } Z2) \\
& = 0
\end{align*}
\(\blacksquare\)

\subsection*{Answer to Question 3.6}
Recall that the indifference condition for reservation wage is given by 
\begin{equation}

\end{equation}
Multiplying both sides with $1 - \beta$ and subtracting $(1 - \beta^{T-t+1}) c$, we have 
\begin{align*}
  (\bar{w}_t - c) (1 - \beta^{T-t+1}) = (\beta^{T-t+1} - \beta ) c + \beta (1 - \beta) \int_{0}^B v_{t+1}(w')dF(w')
\end{align*}
Recall that 
\[
  v_{t+1}(w') = \begin{cases}
  \frac{\bar{w}_{t+1} (1 - \beta^{T-t})}{1 - \beta} &  \text{if } w' \leq \bar{w}_{t+1},\\
  \frac{w' (1 - \beta^{T-t})}{1 - \beta}  & \text{if }  w' > \bar{w}_{t+1}
\end{cases}
\]

Therefore, we have 
\begin{align*}
  (\bar{w}_t - c) (1 - \beta^{T-t+1}) & = - \beta (1 - \beta^{T-t}) c + \beta (1 - \beta^{T-t}) \bar{w}_{t+1} \int_{0}^{\bar{w}_{t+1}}dF(w') + \beta (1 - \beta^{T-t}) \int_{\bar{w}_{t+1}}^B w' dF(w') \\

  & = \beta (1 - \beta^{T-t}) [\mathbb{E}[w'] - c + \int_{0}^{\bar{w}_{t+1}} (\bar{w}_{t+1} - w') dF(w')]
\end{align*}
Applying Integral by Part, $\int_{0}^{\bar{w}_{t+1}} (\bar{w}_{t+1} - w') dF(w') = \int_0^{\bar{w}_{t+1}} F(w')dw'$.
Therefore, we have
\[
\bar{w}_t = c + \frac{\beta (1 -\beta^{T-t})}{1 - \beta^{T-t+1}} \bigg[ \mathbb{E}[w'] - c + \int_{0}^{\bar{w}_{t+1}}F(w')dw' \bigg]
\]
Now denote $\frac{\beta (1 -\beta^{T-t})}{1 - \beta^{T-t+1}} = \tilde{\beta}_t$, then 
$g_{t+1}(w) = c + \tilde{\beta}_t \bigg[ \mathbb{E}[w'] - c + \int_{0}^{\bar{w}_{t+1}}F(w')dw' \bigg]$.
It follows that 
\begin{itemize}

\subsection*{Answer to Question 3.12}
Recall that the indifference condition for the reservation wage in the infinite horizon is given by
\begin{align*}
  \frac{\bar{w}}{1 - \beta} & = c + \beta \bigg[\int_{0}^{\bar{w}} \frac{\bar{w}}{1 - \beta} dF(w') + \int_{\bar{w}}^B \frac{w'}{1 - \beta} dF(w') \bigg] \\
  & = c + \beta \bigg[\frac{\bar{w}}{1 - \beta} + \int_{\bar{w}}^B \frac{w' - \bar{w}}{1 - \beta} dF(w') \bigg] \\
\end{align*}
$\Rightarrow$
\begin{align*}
  \bar{w} - c = \frac{\beta}{1 - \beta}  \int_{\bar{w}}^B (w' - \bar{w}) dF(w') \\
\end{align*}
Define a function $g(w) = w - c - \frac{\beta}{1 - \beta}  \int_{w}^B (w' - w) dF(w')$, and then 
apply discretization listed above, we have
\begin{equation}
  \label{eqn:discretized_g}
  \tilde{g}(w) = w - c - \frac{\beta}{1 - \beta} \sum_{\{i: w_i > w \}} (w_i - w) f_i
\end{equation}
To solve the equation above, I apply \texttt{Roots} package in \texttt{Julia}.
