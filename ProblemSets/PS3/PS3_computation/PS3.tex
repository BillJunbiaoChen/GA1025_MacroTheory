\subsection*{Problem 1.1}
The sequence problem of the Cass-Koopmans model is given by 
\begin{align*}
    & V^*(k_0) = \max_{\{k_{t+1}\}_0^\infty} \sum_{t=0}^\infty \beta^t \frac{c_t^{1 - \gamma} - 1}{1 - \gamma} \\
    \text{subject to } \quad & k_{t+1} = (1 - \delta) k_t + A k_t^\alpha - c_t \\
    & k_0 \text{ is given }
\end{align*}
After substituting the resource constraints, we have 
\[
 V^*(k_0) = \max_{\{k_{t+1}\}_0^\infty} \sum_{t=0}^\infty \beta^t \frac{((1 - \delta) k_t + A k_t^\alpha - k_{t+1})^{1 - \gamma} - 1}{1 - \gamma}
\]
Take the derivative w.r.t. $k_{t+1}$, we have 
\[
\beta^t \bigg[ (1-\delta)k_t + Ak_t^\alpha - k_{t+1} \bigg]^{-\gamma} (-1) + \beta^{t+1} 
 \bigg[ (1-\delta)k_{t+1} + Ak_{t+1}^\alpha - k_{t+2} \bigg]^{-\gamma} \bigg(1 - \delta + A \alpha k_{t+1}^{\alpha - 1}\bigg) = 0
\]
After simplification, we have 
\[
\bigg[ (1-\delta)k_t + Ak_t^\alpha - k_{t+1} \bigg]^{-\gamma} = \beta
 \bigg[ (1-\delta)k_{t+1} + Ak_{t+1}^\alpha - k_{t+2} \bigg]^{-\gamma} \bigg(1 - \delta + A \alpha k_{t+1}^{\alpha - 1}\bigg)
\]

In the steady state, $k_t = k_{t+1} = k_{t+2} = \bar{k}$, as a result, we can compute the ss $\bar{k}$ as follows.
\[
\bar{k} = \left(\frac{1 - \beta + \delta \beta}{A \alpha \beta} \right)^{\frac{1}{\alpha - 1}}
\]

After substituting $c_t$ back to the Euler equation, we have 
\[
c_t^{-\gamma} = \beta (1 - \delta + A \alpha k_{t+1}^{\alpha - 1}) c_{t+1}^{-\gamma}
\]
The equation above implies the behavior of consumption smoothing, 
where the agent equalizes the marginal utility of spending today (LHS) and 
the marginal utility of spending tomorrow (RHS).

\section*{Shooting Algorithm}
\subsection*{Problem 1.6}
The reward function is given by
\[
F(x, y) = \frac{((1 - \delta) x + A x^\alpha - y)^{1 - \gamma} - 1}{1 - \gamma}
\]
Therefore, the Bellman Equation is 
\[
V(k) = \max_{k' \in \Gamma(k)} \quad F(k, k') + \beta V(k')
\]
It follows that the Euler equation (using $k_t, k_{t+1}, k_{t+2}$) is 
\[
\left( (1 - \delta)k_t + A k_t^\alpha - k_{t+1} \right)^{-\gamma} = \beta \left( (1 - \delta)k_{t+1} + A k_{t+1}^\alpha - k_{t+2} \right)^{-\gamma} \left( 1 - \delta + A \alpha k_{t+1}^{\alpha - 1} \right)
\]
and the transversality condition is
\[
\lim_{t \rightarrow \infty} \beta^t \left( (1 - \delta)k_t + A k_t^\alpha - k_{t+1} \right)^{-\gamma} k_{t+1} = 0
\]

\section*{Second Order Difference Equation}
\subsection*{Problem 1.10}
The objective function for the sequence problem in the adjustment model is given by
\begin{align*}
    & V^*(x_0) = \max_{\{x_{t+1}\}_0^\infty \in \mathbb{R} } \sum_{t=0}^\infty \beta^t \bigg[- \frac{\phi}{2} x_{t+1}^2 - \frac{\kappa}{2}(x_{t+1} -  x_t)^2 \bigg]
\end{align*}
Take the first-order condition with respect to $x_{t+1}$, we have 
\[
\beta^t \left[-\phi x_{t+1} - \kappa (x_{t+1} - x_t)\right] + \beta^{t+1} \kappa(x_{t+2} - x_{t+1}) = 0
\]

After simplication, we have 
\begin{equation}
\label{eqn:adjustment_costs_model}
\beta \kappa(x_{t+2} - x_{t+1}) =\phi x_{t+1} + \kappa (x_{t+1} - x_t) 
\end{equation}

According to the Definition, the transversality condition is given by 
\[
\lim_{t \rightarrow \infty}  \beta^t \kappa \left( x_{t+1} - x_t \right) x_t = 0 
\]

\subsection*{Answer to Question 1.12}
Therefore, the general solution to the problem is given by 
\[
x_t = A \bigg(\frac{\left( \frac{\phi + \kappa }{\beta \kappa } + 1 \right) -  \sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}} }{2} \bigg)^t + B \bigg(\frac{\left( \frac{\phi + \kappa }{\beta \kappa } + 1 \right) +  \sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}} }{2} \bigg)^t 
\]


\noindent Let $m_1 < m_2$, since $m_1 \cdot m_2 = \frac{1}{\beta}$, we have
\begin{align*}
& m_1 < \frac{1}{\beta m_1} \\
\Rightarrow & \quad m_1 < \beta^{-1/2} \quad \text{ and } \quad m_2 > \beta^{-1/2}
\end{align*}
Then, we can express $m_1$ and $m_2$ as 
\begin{align*}
    m_1 & = \beta^{-1/2} \epsilon, \quad \epsilon \in (0,1) \\ 
    m_2 & = \beta^{-1/2} \delta, \quad \delta > 1)  \\ 
\end{align*}
Substituting $x_t$ with $A ( \beta^{-1/2} \epsilon)^t + B (\beta^{-1/2} \delta)^t$, 
the Transversality condition becomes 
\begin{align*}
    & \lim_{t \rightarrow \infty} \beta^t \left[  A ( \beta^{-1/2} \epsilon)^{t+1} + B (\beta^{-1/2} \delta)^{t+1} - A ( \beta^{-1/2} \epsilon)^t - B (\beta^{-1/2} \delta)^t\right] \cdot \left[  A ( \beta^{-1/2} \epsilon)^t + B (\beta^{-1/2} \delta)^t \right] = \\ 
& \lim_{t \rightarrow \infty} \beta^t \left[  A ( \beta^{-1/2} \epsilon)^{t}\left( \beta^{-1/2} \epsilon - 1 \right) + B (\beta^{-1/2} \delta)^{t} \left(\beta^{-1/2} \delta - 1 \right) \right] \cdot \left[  A ( \beta^{-1/2} \epsilon)^t + B (\beta^{-1/2} \delta)^t \right]  \leq \\ 
& \lim_{t \rightarrow \infty} A^2 \epsilon^{2t} \left( \beta^{-1/2} \epsilon - 1 \right) + B^2 \delta^{2t} \left(\beta^{-1/2} \delta - 1 \right) 
\end{align*}
Because $\delta > 1$, $\delta^{2t} \rightarrow \infty$, therefore, this equation converges to zero if $B = 0$,
where $B$ is the coefficient of the larger root.

\subsection*{Answer to Question 1.14}
From 1.13, we have
\[
x_t = x_0 \bigg(\frac{\left( \frac{\phi + \kappa }{\beta \kappa } + 1 \right) -  \sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}} }{2} \bigg)^t 
\]

Take the first-order condition with respect to $\kappa$ for $\phi$, 
\begin{align*}
\frac{\partial x_t}{\partial \kappa} = \frac{x_0}{2^t} t \left(\frac{\phi + \kappa }{\beta \kappa } + 1 -  \sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}} \right)^{t-1} \frac{\phi}{\beta \kappa^2} \left(\frac{1}{\sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}}} - 1\right)
\end{align*}

and
\[
\frac{\partial x_t}{\partial \phi} = \frac{x_0}{2^t} t \left(\frac{\phi + \kappa }{\beta \kappa } + 1 -  \sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}} \right)^{t-1} \frac{1}{\beta \kappa} \left(1 - \frac{1}{\sqrt{\left(\frac{\phi + \kappa }{\beta \kappa } + 1 \right)^2 -\frac{4}{\beta}}} \right)
\]

Intuitively, as $\kappa$ increases, the adjustment cost increases, and thus the agent adjusts $x_t$ more slowly.
Conversely, as $\phi$ increases, the marginal cost of holding $x_t$ increases, and thus the agent adjusts $x_t$ more quickly.


\section*{Guess and Verify}
\subsection*{Answer to Question 1.15}
Incorporting the budget constraint, we can rewrite the model as follows.

\begin{align*}
& \max_{k_{t+1} \in \Gamma(k_t) } \sum_{t=0}^\infty \beta^t U(G(k_t) + (1 - \delta) k_t - k_{t+1}) \\ 
\text {where } & \Gamma(k_t) = [0, G(k_t) + (1 - \delta) k_t]
\end{align*}
Therefore, the Bellman equation is given by 
\[
V(k) = \max_{k'} \quad  U(G(k) + (1 - \delta) k - k') + \beta V(k')
\]


\subsection*{Answer to Question 1.16}
If $U(c) = \log (c)$, $G(k) = k^\alpha$, and $\delta = 1$, and 
we ``guess'' the value function is $V(k) = a + b \log k$, 
the Bellman equation formulated above becomes 
\[
a + b \log k = \max_{k'} \quad \log(k^\alpha - k') + \beta (a + b \log k')
\]
Then,
\[
[k'] = - \frac{1}{k^\alpha - k'} + \beta \frac{b}{k'} = 0
\]
As a result, the policy function is 
\[
k' = \frac{\beta b k^\alpha}{1 + \beta b}
\]

Substituting $k' = \frac{\beta b k^\alpha}{1 + \beta b}$ back to the Bellman equation, we have 
\[
a + b \log k = \log \left(k^\alpha -  \frac{\beta b k^\alpha}{1 + \beta b} \right) + \beta a + \beta \log \left( \frac{\beta b k^\alpha}{1 + \beta b} \right)
\]
By comparing coefficients, we have 
\begin{align*}
    b & = \frac{\alpha}{1 - \alpha \beta} \\ 
    a & = \frac{1}{1 - \beta} \bigg[ \frac{1}{1 - \alpha \beta} \log (1 - \alpha \beta) + \frac{\alpha \beta}{1 - \alpha \beta} \log \frac{\alpha \beta}{1 - \alpha \beta} \bigg]
\end{align*}

In a stochastic environment of a fair dice, the filtered probability space is given by 
\begin{itemize}
    \item $\Omega := \{1, 2, 3, 4, 5, 6 \}^\infty$ denotes the sample space.
    \item $\mathcal{F} := 2^\Omega$
    \item $\{\mathcal{F}_t\}_t^\infty$ is defined as follows:
    \begin{align*}
        \mathcal{F}_1 & = 2^{\{1, 2, 3, 4, 5, 6\}} \\
        \dots \\
        \mathcal{F}_t & = 2^{\{1, 2, 3, 4, 5, 6\}^t} \\
        \dots \\

    \end{align*}
    with $\mathcal{F}_{t} \subseteq \mathcal{F}_{t+s}$ for any $s \geq 0$. 
    Economically, $\mathcal{F}_{t}$ contains the information up to time $t$.
    \item $P = \frac{1}{2^\infty}$
\end{itemize}
