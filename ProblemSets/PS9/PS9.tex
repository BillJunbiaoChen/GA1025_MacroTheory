\section{Exercise 9.6 CRRA utility}
\subsection{a. Answer}
\noindent \textbf{Definition} A competitive equilibrium is given by 
an allocation, and a price system such that 
\begin{enumerate}
    \item The allocation solves the consumer's problem.
    \item Market clears: $c_t = d_t$.
\end{enumerate}

\subsection{b. Answer}
With time-0 trading, we have the FOCs as follows.
\[
\beta^t \pi_t(s^t) c_t(s^t)^{-\gamma} = q_t^0(s^t)
\]
Using market clearing condition, we have 
\[
\beta^t \pi_t(s^t) d_t(s^t)^{-\gamma} = q_t^0(s^t)
\]
Therefore, given 
$\tilde{s}^5 = (\lambda_1, ..., \lambda_5) = 
(0.97, 0.97, 1.03, 0.97, 1.03)$, 
we have 
\[
q_5^0(s^t = \tilde{s}^5) = 0.95^5 \times 
(0.97 \times  0.97 \times  1.03 \times  0.97 \times 1.03)^{-2} \times 
(0.8 \times 0.8 \times 0.2 \times 0.1 \times 0.2) = 2.11 \times 10^{-3}
\]

\subsection{c. Answer}
\[
q_5^0(s^t = \tilde{s}^5) = 0.95^5 \times 
(1.03^4 \times 0.97)^{-2} \times 
(0.2 \times 0.9 \times 0.9 \times 0.9 \times 0.1) = 2.11 \times 10^{-3} = 9.47\times 10^{-3}
\]

\subsection{d. Answer}
Since $q^0_t(s^t) = \beta^t \pi_t(s^t) d_t(s^t)^{-\gamma}$, the price at time 0 
of a claim on the entire endowment sequence is given by 
\begin{align*}
p & = \sum_{t=0}^\infty \sum_{s^t} \beta^t \pi_t(s^t) d_t(s^t)^{-\gamma} d_t(s^t) \\ 
    & = \sum_{t=0}^\infty \sum_{s^t} \beta^t \pi_t(s^t) d_t(s^t)^{1-\gamma} 
\end{align*}

\subsection{e. Answer}
To facilitate the following computation, let's Bellmanize the 
the formula for the price of a claim on the entire endowment.
Define 
\[
P(s_0) := \sum_{t=0}^\infty \sum_{s^t} \beta^t \pi_t(s^t) d_t(s^t)^{1-\gamma} = \sum_{t=0}^\infty \sum_{s^t|s_0} \beta^t \pi(s^t|s_0) d_t(s^t)^{1-\gamma},
\]

Also, we can separate the time-0 term as follows 
\begin{align*}
    P(s_0) & = \sum_{t=0}^\infty \sum_{s^t} \beta^t \pi(s^t) d_t(s^t)^{1-\gamma} \\
    & = \pi(s_0) d_0^{1 - \gamma} + \sum_{t=1}^\infty \sum_{s^t} \beta^t \pi(s^t) d_t(s^t)^{1-\gamma} \\
    & = \pi(s_0) d_0^{1 - \gamma} + \sum_{s_1}\sum_{t=1}^\infty \sum_{s^t|s_1} \beta^t \pi(s^t|s_1) \pi(s_1|s_0) d_t(s^t)^{1-\gamma} \\ 
    & = 1 + \beta \sum_{s_1} \pi(s_1|s_0) \sum_{t=1}^\infty \sum_{s^t|s_1} \beta^{t-1} \pi(s^t|s_1)  d_t(s^t)^{1-\gamma} \\
    & = 1 + \beta \sum_{s_1} \pi(s_1|s_0) P(s_1)
\end{align*}
Therefore, we have 
\begin{align*}
    P_L & = 1 + 0.95 \times (0.8 P_L + 0.1 P_H) \\
    P_H & = 1 + 0.95 \times (0.9 P_H + 0.2 P_L)
\end{align*}

\subsection{f. Answer}

\noindent \textbf{Definition}
A recursive competitive equilibrium is given by 
an initial wealth $a_0$, a set of borrowing limits 
$A(s)$, a pricing kernel $Q(s'|s)$, 
a set of value functions $\{V(a, s) \}$, and 
decision rules $\{c^i(s), {a'}(s) \}$ such that 
\begin{enumerate}
    \item The state-by-state borrowing constraints satisfy the recursion: $  A(s) = y(s) + \sum_{s'} Q(s'|s)  A(s')$.
    \item Given $a^1_0$, $A^1(s)$, and $Q(s'|s)$, the value function and decision rules solve the consumers' problems.
    \item For all realizations of $\{s_t \}_{t=0}^\infty$, the consumption and asset portfolios $\{ c^i_t, \{{{a_{t+1}'}}(s')\}_{s'}\}_t$ 
    satisfy 
    \begin{align*}
        c_t & = d(s_t)  \\
        {a'_{t+1}} & = 0
    \end{align*}
    for all $t$ and $s'$.
\end{enumerate}

\subsection{g. Answer}
The state-by-state borrowing constraints is given by 
\[
A(s) = y(s) + \sum_{s'} Q(s'|s)  A(s')
\]

\subsection{h. Answer}
Per the Benveniste-Scheinkman formula, we have 
\[
Q(s_{t+1}|s_t) = \frac{\beta \pi(s_{t+1}| s_t) \left[c_{t+1}(s^{t+1})\right]^{-\gamma}}{c_{t}(s^t)^{-\gamma}}
\]
Plugging in the market clearing condition, we have 
\begin{align*}
    Q(s_{t+1}|s_t) & = \frac{\beta \pi(s_{t+1}| s_t) \left[d_{t+1}(s^{t+1})\right]^{-\gamma}}{d_{t}(s^t)^{-\gamma}} \\ 
    & = \beta \pi(s_{t+1}| s_t) \lambda_{t+1}^{-\gamma}
\end{align*}

\subsection{i. Answer}
Using No-Arbitrage condition and the recursion of the pricing kernel, we have 
\begin{align*}
    p & = \sum_{s_{t+1}} \sum_{s_t} Q(s_{t+2}|s_{t+1}) Q(s_{t+1}|s_t) \\ 
        & = \sum_{s_{t+1}} \sum_{s_t} \beta^2 \pi(s_{t+2}| s_t) ( \lambda_{t+2} \lambda_{t+1})^{-\gamma}
\end{align*}



\section{Exercise 9.7 Log utility}
\subsection{a. Answer} 
Denote the transition matrix $P$, we have 
\[
\pi_{t+1}' = \pi_t' P
\]
It follows that 
\[
\pi_t' = \pi_0' P^t.
\]

\subsection{b. Answer} 
The Largrangian is given by 
\[
\mathcal{L} =  \sum_{t=0}^\infty \sum_{s^t} \beta^t  pi_t(s^t)  \bigg(\theta \ln \left[ c_t^1 (s^t) \right] + (1 - \theta) \ln \left[ c_t^2 (s^t) \right] \bigg) 
+ \mu \sum_{t=0}^\infty \sum_{s^t} \bigg(Y_t(s_t) - c_t^1 (s^t) - c_t^2 (s^t) \bigg)
\]
The FOCs are:
\begin{align*}
    [c^1_t(s^t)]: \quad \mu & = \frac{\theta \beta^t \pi_t(s^t) }{c_t^1(s^t)} \\
    [c^2_t(s^t)]: \quad \mu & = \frac{(1 - \theta) \beta^t \pi_t(s^t) }{c_t^2(s^t)}  \\
\end{align*}
Applysing the market clearing condition: $Y_t = 1 + s_t = c_t^1(s^t) + c_t^2(s^t)$, we have 
\[
\frac{\theta  }{c_t^1(s^t)} = \frac{1 - \theta  }{1+ s_t - c_t^1(s^t)} 
\]
Therefore, the solution is given by 
\begin{align*}
    c^1_t(s^t) & =  \theta (1 + s_t) \\
    c^2_t(s^t) & =  (1 - \theta) (1 + s_t) \\
\end{align*}


\subsection{c. Answer} 
The competitive equilibrium with time-0 trading is given by 
an allocation $\{c_t^1(s^t), c_t^2(s^t)\}$ and a price system $q_t^0(s^t)$, such that 
\begin{enumerate}
    \item the allocation solves each consumer's problem,
    \item market clears: ${c_t(s^t)}^1 + {c_t(s^t)}^2 = {y_t(s^t)}^1 + {y_t(s^t)}^2 = Y_t, \quad \forall t, s^t$. 
\end{enumerate}


\subsection{d. Answer} 
The FOCs for agents 1 and 2 imply that
\begin{align*}
    \beta^t \pi_t(s^t) [c_t^1(s^t)]^{-1} &= \mu^1 q_t^0(s^t) \\
    [c_0^1(s_0)]^{-1} &= \mu^1 
\end{align*}

\begin{align*}
    \beta^t \pi_t(s^t) [c_t^2(s^t)]^{-1} &= \mu^2 q_t^0(s^t) \\
    [c_0^2(s_0)]^{-1} &= \mu^2
\end{align*}
where I noramlize $q_0^0(s_0) = 1$ and use the facts that $\beta^0 =1$, and $\pi_0(s_0)=1$. 

Adding these equations together, we have 
\[
\beta^t \pi_t(s^t)Y_0(s_0) = Y_t(s_t) q_t^0(s^t)
\]
Therefore, we have
\[
q_t^0(s^t) = \frac{\beta^t \pi_t(s^t)Y_0(s_0)}{Y_t(s_t) } = \frac{\beta^t \pi_t(s^t)(1+s_0)}{1 + s_t} 
\]


\subsection{e. Answer} 
From (d), we have 
\[
\beta^t \pi_t(s^t) [c_t^1(s^t)]^{-1} = \mu^1 \frac{\beta^t \pi_t(s^t)(1+s_0)}{1 + s_t} 
\]
Massaging terms on both sides, we have 
\[
c_t^1(s^t) = \frac{1}{\mu^1 (1 + s_0)} (1 + s_t),
\]
which coincides with the Pareto solution if $\theta = \frac{1}{\mu^1 (1 + s_0)}$.

\noindent \textbf{Remark:} The First Welfare Theorem implies that C.E. is Pareto optimal if there is no externalities.


\subsection{f. Answer} 
In the competitive equilibrium, a price system is determined by inidividuals' MRS (up to some normalization and proper discounting.)


\subsection{g. (Recursive competitive equlibrium) Answer} 
Let $v^i(a,s)$ be the optimal value of agent $i$ with states $a$ (wealth) and $s$ (aggregate state).
The optimal value function satisfies the Bellman equation
\begin{align*}
v^i(a, s) & = \max_{s',a'(s')}\quad \ln c^i(s) + \beta \sum_{s'} \pi(s'|s) v^i(a'(s'),s') \\
\text{subject to } \quad & c^i(s) + \sum_{s'} Q(s'|s) a'(s') \leq y^i(s) + a \\ 
& -a'(s') \leq A^i (s'), \forall s' \in S
\end{align*}
where $A^i(s')$ is state-by-state debt limits given by the following recursive:
\[
A^i(s) = y^i(s) + \sum_{s'} Q(s'|s)  A^i(s'), \quad i = 1, 2
\]

\noindent \textbf{Definition}
A recursive competitive equilibrium is given by 
an initial distribution of wealth $\{a_0^i\}$, a set of borrowing limits 
$A^i(s)$, a pricing kernel $Q(s'|s)$, 
a set of value functions $\{v^i(a, s) \}$, and 
decision rules $\{c^i(s), {a'}^i(s) \}$ such that 
\begin{enumerate}
    \item The state-by-state borrowing constraints satisfy the recursion: $  A^i(s) = y^i(s) + \sum_{s'} Q(s'|s)  A^i(s')$.
    \item For all $i = 1, 2$, given $a^i_0$, $A^i(s)$, and $Q(s'|s)$, the value function and decision rules solve the consumers' problems.
    \item For all realizations of $\{s_t \}_{t=0}^\infty$, the consumption and asset portfolios $\{ \{c^i_t, \{{{a_{t+1}'}^i}(s')\}_{s'} \}_i\}_t$ 
    satisfy 
    \begin{align*}
        c^1_t + c^2_t & = y^1(s_t) + y^2(s_t) \\
        {a'_{t+1}}^1 + {a'_{t+1}}^2 & = 0
    \end{align*}
    for all $t$ and $s'$.
    \item The initial wealth distribution satisfies $a_0^1 + a_0^2 = 0$.
\end{enumerate}


\subsection{h. Answer} 
Using the FOCs of the problem and a Benveniste-Scheinkman formula, we have 
\[
Q(s_{t+1}|s_t) = \frac{\beta [c^i_{t+1}(s_{t+1})]^{-1} \pi(s_{t+1} | s_t)
    }{ [c^i(s_t)]^{-1}}, \quad \forall i = 1, 2
\]
It follows that 
\[
\frac{c^1(s_t)}{c^1_{t+1}(s_{t+1})} = \frac{c^2(s_t)}{c^2_{t+1}(s_{t+1})} = \frac{c^1(s_t) + c^2(s_t)}{c^1_{t+1}(s_{t+1}) + c^2_{t+1}(s_{t+1})}
\]
Therefore, we have 
\[
Q(s_{t+1}|s_t) = \frac{\beta Y(s_t) \pi(s_{t+1} | s_t)}{ Y(s_{t+1})}
\]
Given the parameterization of $\beta = 0.95$ and 
$P = \begin{bmatrix}
    0.8 & 0.2 \\ 0.3 & 0.7
\end{bmatrix} $, we have 

\begin{table}[h!]
\centering
\begin{tabular}{ccc}
\hline
$s_t$ & $s_{t+1}$ & $Q(s_{t+1} | s_t)$ \\ \hline
0     & 0         & $0.95 \cdot 0.8$ \\ 
0     & 1         & $0.95 \cdot \frac{1}{2} \cdot 0.2$ \\  
1     & 0         & $0.95 \cdot \frac{2}{1} \cdot 0.3$ \\  
1     & 1         & $0.95 \cdot 0.7$ \\ \hline

\end{table}

\subsection{i. Answer} 
Therefore, using No-Arbitrage condition, 
the prices of one-period risk-free bond are given by 
\[
p(s) = \sum_{s_{t+1}} Q(s_{t+1}|s_t = s)
\]
where $s = 0$ or $1$.

\subsection{j and k. Answers} 
The $j$-periods ahead Arrow securities satisfy the following recursion 
\[
Q_j(s_{t+j}|s_t) = \sum_{s_{t+1}} Q_{j-1}(s_{t+j}|s_{t+1}) Q_1(s_{t+1}|s_t)
\]
where $Q_1(s_{t+1}|s_t) := Q(s_{t+1}|s_t)$.

Using No-arbitrage condition, the price of j-period-ahead risk-free bond is 
\[
p^j(s) = \sum_{s_{t+j}} Q_j(s_{t+j}|s_t = s)
\]
Due to symmetry, we have 
\[
c^1_t(s^t) = c^2_t(s^t) = 1/2, \forall s^t
\]

\subsection{Part II: e. Answer} 
Under the assumption of diverse beliefs, we have 
\[
\pi^2_t(s^t)' = \pi_0 \hat{P}^t
\]

\subsection{f. Answer} 
Denote the Pareto weights by 
$\lambda_1 := \theta$ and $\lambda_2 := 1 - \theta$.
Then, we have 
\[
\mathcal{L} = \sum_i \lambda_i \sum_{t=0}^\infty \sum_t \beta^t \ln [c_t^i(s^t)] \pi^i_t(s^t) 
- \mu \bigg(\sum_i \sum_{t=0}^\infty \sum_t y_t^i(s^t) - c_t^i(s^t) \bigg)
\]
The FOCs are 
\begin{align*}
    \lambda_1 \beta^t \pi^1_t(s^t) [c_t^1(s^t)]^{-1} & = \mu \\
    \lambda_2 \beta^t \pi^2_t(s^t) [c_t^2(s^t)]^{-1} & = \mu
\end{align*}
It follows that 
\[
\frac{\pi^1_t(s^t) c_t^2(s^t)
    }{
        \pi^t_t(s^t) c_t^1(s^t)
    } = \frac{\lambda_2}{\lambda_1}
\]
Since $c_t^1(s^t) + c_t^2(s^t) = 1$, we have 
\[
\frac{\pi^1_t(s^t) (1 - c_t^1(s^t))
    }{
        \pi^2_t(s^t) c_t^1(s^t)
    } = \frac{\lambda_2}{\lambda_1}
\]
Then, 
\[
c_t^1(s^t) = \frac{\theta \pi^1_t(s^t)}{\theta \pi^1_t(s^t) + (1- \theta) \pi^2_t(s^t)}, \qquad
c_t^2(s^t) = \frac{(1- \theta) \pi^2_t(s^t)}{\theta \pi^1_t(s^t) + (1- \theta) \pi^2_t(s^t)}
\]

\subsection{g. Answer} 
\noindent \textbf{Definition}
An equilibrium with time-0 trading of a complete set of Arrow-Debreu $s^t$-contingent securities is given by 
an allocation, and a price system such that 
\begin{enumerate}
    \item Given the price system, the allocation solves consumers 1 and 2's problems,
    \item Market clears $c^1_t(s^t) + c^2_t(s^t) = y^1_t(s^t) + y^2_t(s^t), \quad \forall s^t$.
\end{enumerate}


\subsection{h. Answer} 
The Largrangian is given by 
\[
\mathcal{L}^i = \sum_{t=0}^\infty \sum_t \beta^t \ln [c_t^i(s^t)] \pi^i_t(s^t) 
- \mu^i \bigg[\sum_{t=0}^\infty \sum_t q^0_t(s^t) \bigg(y_t^i(s^t) - c_t^i(s^t) \bigg) \bigg]
\]
The FOCs are 
\begin{align*}
   \beta^t \pi^i_t(s^t) [c_t^i(s^t)]^{-1} & = \mu^i q_t^0(s^t) \\
   [c_0^i(s_0)]^{-1} & = \mu^i
\end{align*}
It follows that 
\[
\beta^t \pi^i_t(s^t) c_0^i(s_0) = q_t^0(s^t) c_t^i(s^t)
\]
\noindent $\Rightarrow$
\[
\beta^t \bigg(\pi^1_t(s^t) c_0^1(s_0) + \pi^2_t(s^t) c_0^2(s_0) \bigg) = q_t^0(s^t)
\]
Notice that after period 1, the states become deterministic.
That is, $s_t = 0$ or $2$ for $t \geq 2$.
Therefore, we have 
\[
\frac{q^0_{t+1}(s^{t+1})
}{q^0_{t}(s^{t})} = \frac{\beta \pi^i (s_{t+1}| s_t) c^i_t(s^t)}{c^i_{t+1}(s^{t+1})}
\]
As I said previously, there is no diverse beliefs and uncertainty after period 1.
Hence, $\pi^1(s_{t+1}|s_t) = \pi^2(s_{t+1}|s_t) = 1$.
Then, we have 
\[
q^0_{t+1}(s^{t+1}) = \beta q^0_{t}(s^{t}) = \beta^t q^0_{1}(s^1)
\]
